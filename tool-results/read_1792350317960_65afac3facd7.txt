    54→/* ---------------- clipboard (module-level, shared across windows) -------- */
    55→
    56→interface ClipboardData {
    57→  ids: string[];
    58→  mode: "copy" | "cut";
    59→}
    60→
    61→let clipboard: ClipboardData | null = null;
    62→const clipboardSubs = new Set<() => void>();
    63→
    64→function setClipboardData(next: ClipboardData | null) {
    65→  clipboard = next;
    66→  for (const fn of clipboardSubs) fn();
    67→}
    68→
    69→function useClipboard(): ClipboardData | null {
    70→  const [value, setValue] = useState<ClipboardData | null>(clipboard);
    71→  useEffect(() => {
    72→    const fn = () => setValue(clipboard);
    73→    clipboardSubs.add(fn);
    74→    return () => {
    75→      clipboardSubs.delete(fn);
    76→    };
    77→  }, []);
    78→  return value;
    79→}

   151→    setNav((n) => (n.index > 0 ? { ...n, index: n.index - 1 } : n));
   152→    setSelectedId(null);
   153→  }, []);
   154→

   156→    setNav((n) => (n.index < n.stack.length - 1 ? { ...n, index: n.index + 1 } : n));

   158→  }, []);

   167→  const refreshStorage = useCallback(async () => {

   173→    }
   174→  }, []);

   176→  useEffect(() => {
   177→    navigator.storage
   178→      ?.estimate?.()

   180→      .catch(() => setStorage(null));
   181→  }, []);
   182→

   189→    const win = st.windows.find((w) => w.id === windowId);

   203→      const node = ns[target];

   205→        navigate(node.id);

   236→    });

   306→            parentId,
   307→            name: file.name,
   308→            type: "file",
   309→            binary,
   310→            mime: file.type || undefined,
   311→            autoRename: true,

   314→        }
   315→      } catch (err) {
   316→        os.notify({
   317→          title: "Files",
   318→          body: `Could not import "${file.name}" — ${err instanceof Error ? err.message : "unknown error"}`,
   319→        });
   320→      }
   321→    }
   322→  }, []);

   370→          onClick: () => {

   389→          label: "Delete",

   453→    },

   474→      setDragOverId((cur) => (cur === node.id ? null : cur));
   475→    },
   476→    onDrop: (node, e) => {
   477→      e.preventDefault();
   478→      e.stopPropagation();
   479→      draggingIdRef.current = null;
   480→      setDragOverId(null);
   481→      if (e.dataTransfer.files.length > 0) {
   482→        void importFiles(Array.from(e.dataTransfer.files), node.id);
   483→        return;
   484→      }
   485→      const dragged = e.dataTransfer.getData(NODE_MIME);
   486→      if (dragged && dragged !== node.id) {
   487→        const err = useFS.getState().move(dragged, node.id);
   488→        if (err) useOS.getState().notify({ title: "Files", body: err });
   489→      }
   490→    },
   491→    onDragEnd: () => {
   492→      draggingIdRef.current = null;
   493→      setDragOverId(null);
   494→    },
   495→  };
   496→
   497→  const handleContentDrop = useCallback(
   498→    (e: ReactDragEvent) => {
   499→      e.preventDefault();
   500→      draggingIdRef.current = null;
   501→      setDragOverId(null);
   502→      if (e.dataTransfer.files.length > 0) {
   503→        void importFiles(Array.from(e.dataTransfer.files), dirId);
   504→        return;
   505→      }
   506→      const dragged = e.dataTransfer.getData(NODE_MIME);
   507→      if (dragged && dragged !== dirId) {
   508→        const err = useFS.getState().move(dragged, dirId);
   509→        if (err) useOS.getState().notify({ title: "Files", body: err });
   510→      }
   511→    },
   512→    [dirId, importFiles]
   513→  );
   514→
   515→  /* ---------- keyboard ---------- */
   516→
   517→  useEffect(() => {
   518→    const onKey = (e: KeyboardEvent) => {
   519→      const el = document.activeElement as HTMLElement | null;
   520→      const tag = el?.tagName ?? "";
   521→      const typing =
   522→        tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || (el?.isContentEditable ?? false);
   523→      if (e.key === "Escape") {
   524→        if (modal) {
   525→          setModal(null);
   526→          return;
   527→        }
   528→        if (!typing) setSelectedId(null);
   529→        return;
   530→      }
   531→      if (modal || typing || tag === "BUTTON" || !isFocused || e.metaKey || e.ctrlKey || e.altKey) {
   532→        return;
   533→      }
   534→      const node = selectedId ? nodes[selectedId] : undefined;
   535→      if (!node) return;
   536→      if (e.key === "Delete") {
   537→        e.preventDefault();
   538→        setModal({ kind: "delete", id: node.id });
   539→      } else if (e.key === "F2") {
   540→        e.preventDefault();
   541→        setModal({ kind: "rename", id: node.id });
   542→      } else if (e.key === "Enter") {
   543→        e.preventDefault();
   544→        openNode(node);
   545→      }
   546→    };
   547→    window.addEventListener("keydown", onKey);
   548→    return () => window.removeEventListener("keydown", onKey);
   549→  }, [modal, isFocused, selectedId, nodes, openNode]);
   550→
   551→  /* ---------- render ---------- */
   552→
   553→  const searchQ = query.trim();
   554→  const storagePct =
   555→    storage && storage.quota > 0 ? Math.min(100, Math.round((storage.usage / storage.quota) * 100)) : 0;
   556→
   557→  return (
   558→    <div className="@container flex h-full min-h-0 w-full text-[13px] text-[var(--nx-text)]">
   559→      {/* sidebar */}
   560→      <aside className="nx-scroll @min-[640px]:flex hidden w-[200px] shrink-0 flex-col gap-3 overflow-y-auto border-r border-[var(--nx-border)] p-2">

   562→          <div className="px-2 pb-1 text-[10px] font-semibold uppercase tracking-wider text-[var(--nx-text-dim)]">
